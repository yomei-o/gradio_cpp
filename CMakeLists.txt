cmake_minimum_required(VERSION 3.16)
project(gradio_cpp CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(gradio_cpp INTERFACE)
target_include_directories(gradio_cpp INTERFACE include third_party)
find_package(Threads REQUIRED)
target_link_libraries(gradio_cpp INTERFACE Threads::Threads)
if(WIN32)
  target_link_libraries(gradio_cpp INTERFACE ws2_32)
  target_compile_definitions(gradio_cpp INTERFACE _WIN32_WINNT=0x0A00)
endif()
if(MSVC)
  target_compile_options(gradio_cpp INTERFACE /utf-8 /bigobj)
endif()

foreach(ex hello sepia blocks components extras chat_echo llama_chat)
  add_executable(${ex} examples/${ex}.cpp)
  target_link_libraries(${ex} PRIVATE gradio_cpp)
endforeach()

# Optional: examples/llama_local.cpp links llama.cpp (libllama + libmtmd) directly.
#   cmake -S . -B build -DGRADIO_LLAMA_CPP=path/to/llama.cpp
set(GRADIO_LLAMA_CPP "" CACHE PATH "llama.cpp source tree; builds the llama_local example")
if(GRADIO_LLAMA_CPP)
  set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  add_subdirectory(${GRADIO_LLAMA_CPP} llama.cpp EXCLUDE_FROM_ALL)
  add_executable(llama_local examples/llama_local.cpp)
  target_link_libraries(llama_local PRIVATE gradio_cpp llama mtmd)
endif()
